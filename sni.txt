<script>
    // 1. MODELLO DATI (Mockup per gestire i gruppi e le note)
    const datiIniziali = {
        "Caffetteria": [
            { id: 1, nome: "Caffè Espresso", costo: "0.12 €", prezzo: "1.20 €" },
            { id: 2, nome: "Caffè al Ginseng", costo: "0.45 €", prezzo: "1.80 €" }
        ],
        "Dolce": [
            { id: 3, nome: "Cornetto Crema", costo: "0.60 €", prezzo: "1.40 €" }
        ],
        "Salato": [
            { id: 4, nome: "Focaccia Crudo", costo: "1.80 €", prezzo: "5.50 €" }
        ]
    };

    // 2. FUNZIONE PER FAR COMPARIRE I DATI REALI (Iniezione HTML)
    function renderizzaPagina() {
        const menu = document.getElementById('menu-gruppi');
        const container = document.getElementById('tabelle-container');
        
        Object.keys(datiIniziali).forEach((gruppo, index) => {
            // Crea Bottone Menu
            const btn = document.createElement('button');
            btn.className = `nav-btn ${index === 0 ? 'active' : ''}`;
            btn.innerText = gruppo;
            btn.onclick = (e) => cambiaTab(gruppo, e.target);
            menu.appendChild(btn);

            // Crea Tabella Gruppo
            const table = document.createElement('table');
            table.id = gruppo;
            table.className = `tabella-sezione ${index === 0 ? 'active' : ''}`;
            table.innerHTML = `
                <thead><tr><th>Prodotto</th><th>Costo</th><th>Prezzo</th><th>Note (Database)</th></tr></thead>
                <tbody>
                    ${datiIniziali[gruppo].map(p => `
                        <tr>
                            <td><b>${p.nome}</b></td>
                            <td>${p.costo}</td>
                            <td>${p.prezzo}</td>
                            <td>
                                <input type="text" id="note-${p.id}" class="input-note" value="${localStorage.getItem('nota-'+p.id) || ''}" placeholder="Scrivi nota...">
                                <button class="btn-salva" onclick="salvaNota(${p.id})">💾</button>
                            </td>
                        </tr>
                    `).join('')}
                </tbody>
            `;
            container.appendChild(table);
        });
    }

    // 3. LOGICA DATABASE (Inserimento dati/salvataggio)
    function salvaNota(id) {
        const valore = document.getElementById('note-' + id).value;
        localStorage.setItem('nota-' + id, valore); // Simula il salvataggio nel database
        alert("Nota salvata nel database locale!");
    }

    function cambiaTab(idSezione, btn) {
        document.querySelectorAll('.tabella-sezione').forEach(t => t.classList.remove('active'));
        document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
        document.getElementById(idSezione).classList.add('active');
        btn.classList.add('active');
    }

    // Avvio
    renderizzaPagina();
</script>
