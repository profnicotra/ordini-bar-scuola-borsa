<script>
    let totale = 0;
    const listaUl = document.getElementById('lista-ordine');
    const totaleSpan = document.getElementById('totale-prezzo');
    const vuotoMsg = document.getElementById('vuoto-msg');

    function aggiungi(nome, prezzo) {
        if (vuotoMsg) vuotoMsg.remove();
        
        const li = document.createElement('li');
        li.className = 'item-ordine';
        li.innerHTML = `
            <div>
                <span style="color: #fff;">${nome}</span>
                <span style="color: #ff8800; margin-left: 10px;">${prezzo.toFixed(2)} €</span>
            </div>
            <i class="fas fa-trash-alt btn-delete" onclick="rimuovi(this, ${prezzo})"></i>
        `;
        listaUl.appendChild(li);
        
        totale += prezzo;
        aggiornaTotale();
    }

    function rimuovi(el, prezzo) {
        el.parentElement.remove();
        totale -= prezzo;
        if (totale < 0) totale = 0;
        aggiornaTotale();
    }

    function aggiornaTotale() {
        totaleSpan.innerText = totale.toFixed(2);
    }

    function conferma() {
        if (totale === 0) return alert("Scegli qualcosa prima!");
        alert("Ordine ricevuto! Iniziamo a preparare la tua colazione.");
        location.reload();
    }
</script>
