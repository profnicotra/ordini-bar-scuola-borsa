<script>
    let totale = 0;
    const PSW_ADMIN = "1234";

    let prodottiMenu = [
        { n: "Caffè Espresso", c: "Caffetteria", co: 0.15, pr: 1.20, pi: 1.00, st: "DISPONIBILE" },
        { n: "Cappuccino", c: "Caffetteria", co: 0.35, pr: 1.80, pi: 1.50, st: "DISPONIBILE" },
        { n: "Cornetto Crema", c: "Dolci", co: 0.50, pr: 1.50, pi: 1.30, st: "DISPONIBILE" },
        { n: "Coca Cola", c: "Bevande", co: 0.60, pr: 2.50, pi: 2.00, st: "DISPONIBILE" },
        { n: "Panino Crudo", c: "Salato", co: 1.80, pr: 4.50, pi: 4.00, st: "DISPONIBILE" }
    ];
    let filtroAttuale = "TUTTI";

    function aggiungiProdotto() {
        const n = document.getElementById('n_in').value;
        const c = document.getElementById('c_in').value || "Varie";
        const co = parseFloat(document.getElementById('co_in').value) || 0;
        const pr = parseFloat(document.getElementById('pr_in').value) || 0;
        const pi = parseFloat(document.getElementById('pi_in').value) || 0;
        if(!n) return;
        prodottiMenu.push({ n, c, co, pr, pi, st: "DISPONIBILE" });
        document.getElementById('n_in').value = "";
        aggiornaInterfaccia();
    }

    function renderMenu() {
        const corpo = document.getElementById('lista-menu');
        corpo.innerHTML = "";
        prodottiMenu.forEach((p, index) => {
            if (filtroAttuale !== "TUTTI" && p.c.toUpperCase() !== filtroAttuale) return;
            const riga = document.createElement('tr');
            const isOk = p.st === "DISPONIBILE";
            riga.className = "info-row " + (isOk ? "" : "riga-no");
            
            riga.onclick = () => { if(isOk) ordina(p.n, p.pr); };
            
            riga.innerHTML = `
                <td class="col-num">${index + 1}</td>
                <td><b style="color:#00c8ff">${p.n}</b></td>
                <td style="color:#777">${p.co.toFixed(2)}€</td>
                <td style="font-weight:bold">${p.pr.toFixed(2)}€</td>
                <td style="color:#00a6ff">${p.pi.toFixed(2)}€</td>
                <td><span style="color:#00a6ff; font-size:0.7em">${p.c}</span></td>
                <td><span class="badge ${isOk ? 'stato-ok' : 'stato-no'}" ondblclick="event.stopPropagation(); cambiaStatoAdmin(${index})">${p.st}</span></td>
                <td><button style="color:#ff4444; background:none; border:none; cursor:pointer" onclick="event.stopPropagation(); eliminaProdotto(${index})">🗑️</button></td>
            `;
            corpo.appendChild(riga);
        });
    }

    function cambiaStatoAdmin(idx) {
        let psw = prompt("Codice ADMIN:");
        if (psw === PSW_ADMIN) {
            prodottiMenu[idx].st = prodottiMenu[idx].st === "DISPONIBILE" ? "ESAURITO" : "DISPONIBILE";
            aggiornaInterfaccia();
        } else { alert("Codice errato!"); }
    }

    function ordina(nome, prezzo) {
        const lista = document.getElementById('lista-ordine');
        const id = 'ord-' + Date.now();
        const voce = document.createElement('div');
        voce.className = "order-item";
        voce.id = id;
        voce.innerHTML = `
            <div class="order-top"><span>${nome}</span><b>${prezzo.toFixed(2)}€</b>
            <button style="color:#ff4444; border:none; background:none; cursor:pointer" onclick="rimuoviDallOrdine('${id}', ${prezzo})">✕</button></div>
            <input type="text" class="allergy-text" placeholder="Note...">`;
        lista.appendChild(voce);
        aggiornaCassa(prezzo);
    }

    function generaScontrino() {
        const items = document.querySelectorAll('.order-item');
        if (items.length === 0) return alert("Vuoto!");
        let testo = "SCUOLA PAOLO BORSA\n----------------------------\n";
        items.forEach(item => {
            testo += `${item.querySelector('span').innerText}: ${item.querySelector('b').innerText}\n`;
        });
        testo += "----------------------------\nTOTALE: " + document.getElementById('totale-ordine').innerText;
        const blob = new Blob([testo], { type: "text/plain" });
        const a = document.createElement("a");
        a.href = URL.createObjectURL(blob);
        a.download = "scontrino.txt";
        a.click();
    }

    function generaListaNumerica() {
        const items = document.querySelectorAll('.order-item');
        if (items.length === 0) return alert("Vuoto!");
        let res = "PRODUZIONE:\n";
        items.forEach((item, i) => { res += `${i+1}. ${item.querySelector('span').innerText}\n`; });
        alert(res);
    }

    function aggiornaInterfaccia() { renderMenu(); renderFiltri(); }
    function renderFiltri() {
        const bar = document.getElementById('filtri');
        const cats = ["TUTTI", ...new Set(prodottiMenu.map(p => p.c.toUpperCase()))];
        bar.innerHTML = cats.map(cat => `<button class="filter-btn ${filtroAttuale === cat ? 'active' : ''}" onclick="filtra('${cat}')">${cat}</button>`).join('');
    }
    function filtra(cat) { filtroAttuale = cat; aggiornaInterfaccia(); }
    function eliminaProdotto(index) { if(confirm("Eliminare?")) { prodottiMenu.splice(index, 1); aggiornaInterfaccia(); } }
    function rimuoviDallOrdine(id, pr) { document.getElementById(id).remove(); aggiornaCassa(-pr); }
    function aggiornaCassa(v) { totale += v; document.getElementById('totale-ordine').innerText = Math.abs(totale).toFixed(2) + " €"; }
    function svuotaComanda() { document.getElementById('lista-ordine').innerHTML = ""; totale = 0; aggiornaCassa(0); }
    window.onload = aggiornaInterfaccia;
</script>
